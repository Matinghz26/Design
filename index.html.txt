<script>
(() => {
  const form = document.getElementById('form');
  const steps = [...document.querySelectorAll('.step')];
  const progress = [...document.querySelectorAll('.progress div')];
  const back = document.getElementById('back');
  const next = document.getElementById('next');
  const navigation = document.getElementById('navigation');
  const summary = document.getElementById('summary');
  const feedback = document.getElementById('feedback');
  let current = 0;

  function val(name) {
    const el = form.elements[name];
    if (!el) return '';
    if (el instanceof RadioNodeList) return el.value || '';
    return el.value || '';
  }

  function spaces() {
    return [...form.querySelectorAll('[name="spaces"]:checked')].map(x => x.value);
  }

  function update() {
    steps.forEach((step, i) => step.hidden = i !== current);
    progress.forEach((item, i) => {
      item.classList.toggle('active', i === current);
      item.classList.toggle('done', i < current);
    });
    back.disabled = current === 0;
    navigation.hidden = current === steps.length - 1;
    next.textContent = current === steps.length - 2
      ? 'ساخت خلاصه ایده ←' : 'مرحله بعد ←';
    if (current === steps.length - 1) renderSummary();
  }

  function validate() {
    if (current !== 0) return true;
    const ok = val('project').trim() !== '' &&
      Number(val('area')) >= 20 && Number(val('area')) <= 2000;
    document.getElementById('error').style.display = ok ? 'none' : 'block';
    return ok;
  }

  function addItem(label, value) {
    const box = document.createElement('div');
    box.className = 'item';
    const small = document.createElement('small');
    small.textContent = label;
    const strong = document.createElement('strong');
    strong.textContent = value || 'انتخاب نشده';
    box.append(small, strong);
    summary.append(box);
  }

  function renderSummary() {
    summary.replaceChildren();
    addItem('نام پروژه', val('project'));
    addItem('نوع خانه', val('type'));
    addItem('زیربنا', val('area') ? val('area') + ' متر مربع' : '');
    addItem('تعداد طبقات', val('floors'));
    addItem('سبک معماری', val('style'));
    addItem('پالت رنگی', val('palette'));
    addItem('اولویت', val('priority'));
    addItem('بودجه', val('budget'));
    addItem('اقلیم یا محل', val('place'));
    addItem('امکانات', spaces().join('، '));
    if (val('notes').trim()) addItem('توضیحات', val('notes').trim());
  }

  function plainText() {
    return [
      'خانه رؤیایی من | خانه‌نگار',
      'نام پروژه: ' + val('project'),
      'نوع خانه: ' + val('type'),
      'زیربنا: ' + val('area') + ' متر مربع',
      'طبقات: ' + val('floors'),
      'سبک: ' + val('style'),
      'رنگ‌ها: ' + val('palette'),
      'اولویت: ' + val('priority'),
      'بودجه: ' + val('budget'),
      'محل: ' + (val('place') || 'مشخص نشده'),
      'امکانات: ' + (spaces().join('، ') || 'انتخاب نشده'),
      'توضیحات: ' + (val('notes') || 'ندارد'),
      '',
      'این خلاصه، ایده اولیه است و نقشه اجرایی محسوب نمی‌شود.'
    ].join('\n');
  }

  async function copyText(text) {
    try {
      await navigator.clipboard.writeText(text);
      return true;
    } catch {
      const textarea = document.createElement('textarea');
      textarea.value = text;
      document.body.append(textarea);
      textarea.select();
      const ok = document.execCommand('copy');
      textarea.remove();
      return ok;
    }
  }

  next.addEventListener('click', () => {
    if (!validate()) return;
    if (current < steps.length - 1) {
      current++;
      update();
      document.querySelector('main').scrollIntoView({behavior:'smooth'});
    }
  });

  back.addEventListener('click', () => {
    if (current > 0) {
      current--;
      update();
      document.querySelector('main').scrollIntoView({behavior:'smooth'});
    }
  });

  document.getElementById('restart').addEventListener('click', () => {
    form.reset();
    current = 0;
    feedback.textContent = '';
    document.getElementById('error').style.display = 'none';
    update();
    document.querySelector('main').scrollIntoView({behavior:'smooth'});
  });

  document.getElementById('copy').addEventListener('click', async () => {
    feedback.textContent = await copyText(plainText())
      ? 'خلاصه با موفقیت کپی شد.'
      : 'کپی خودکار در این مرورگر انجام نشد.';
  });

  document.getElementById('share').addEventListener('click', async () => {
    const text = plainText();
    if (navigator.share) {
      try {
        await navigator.share({title:'خانه رؤیایی من', text});
        feedback.textContent = 'اشتراک‌گذاری انجام شد.';
      } catch (e) {
        if (e.name !== 'AbortError') feedback.textContent = 'اشتراک‌گذاری انجام نشد.';
      }
    } else {
      feedback.textContent = await copyText(text)
        ? 'خلاصه کپی شد؛ می‌توانی آن را برای دیگران بفرستی.'
        : 'اشتراک‌گذاری مستقیم در این مرورگر در دسترس نیست.';
    }
  });

  update();
})();
</script>
